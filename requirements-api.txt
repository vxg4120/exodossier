# Stage 1 public surface (read-only API + MCP server). Python 3.14 venv (.venv).
# All wheels present on 3.14.5; installing these never upgrades the Wave-1/2 scientific pins.
fastapi
uvicorn[standard]
httpx            # also required by starlette's TestClient in the API tests

# Follow-up bridge (api/routers/twoskies.py): SGP4 propagation of the satellite LEO catalog +
# vectorised topocentric geometry against exoplanet-host sightlines. sgp4 ships a cp311-abi3 wheel
# that installs cleanly on Python 3.14; numpy drives the vectorised SatrecArray path (few-seconds
# compute over ~16k objects).
numpy
sgp4

# Postgres driver — the read-only API and the twoskies bridge open psycopg
# connections (api/deps.py, api/queries.py, common/db.py). Ships manylinux +
# aarch64 wheels, so no libpq/build tools are required in the image.
psycopg[binary]

# MCP server (mcp/server.py). Bounded below 2.0: the 2.x SDK renames FastMCP to MCPServer and
# changes the tool API, so mcp/server.py would not import. The old exact pin to 1.14.1 broke on
# 2026-11-04 the other way round: 1.14.1 reached into pydantic's private _typing_extra, which
# pydantic 2.14 no longer exports, so `import mcp` failed wherever the current fastapi/pydantic
# were installed beside it. cryptography (pulled in by 1.15+) ships wheels for 3.13 and 3.14 now.
mcp>=1.30,<2
